# Capture V2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Dobrar leads relevantes por nicho: filtrar relevância antes do detail, multi-termo adaptativo, budget proporcional ao nº de RAs, recaptura incremental (place_id <30d) e telemetria por query.

**Architecture:** O scraper ganha duas fases (`collectCards` → `relevanceScore` → fila de detail). A relevância é fonte única em `src/utils/categories.js` (NICHE_ROOTS já vindo do filtro de sync). O runner central (`capture-advogados-df.js`) ganha budget/tempo, histórico de yield (sidecar meta + colunas novas em `captures`) e segunda passada de termos. Telemetria por query vai para `captures.qs` (jsonb) com fallback tolerante enquanto a migration não existir.

**Tech Stack:** Node 24 CommonJS, Playwright chromium full (headless=new), node:test + assert, Supabase PostgREST puro (`src/utils/sb`), dotenv.

**Spec:** `docs/superpowers/specs/2026-10-18-capture-v2-design.md`

## Global Constraints

- Comandos Windows PowerShell 5.1 (sem bash; `cmd1; if ($?) { cmd2 }`).
- Trocar process.env.QUERIES continua sendo PIPE `|` de query strings (compat serve-web).
- `sbFetch`/`sbFetchAll` de `src/utils/sb`; PostgREST ignora nada — colunas novas só após migration aplicada (fallback obrigatório).
- Playwright chromium FULL com `--headless=new` (headless-shell quebra nesta máquina).
- Commits pequenos; `npm test` deve passar 201+ testes a cada task.

```js
/* plano de captura: [{ra, query}] — Brasília é o DF inteiro: não duplica o termo */
function buildPlan(term, opts = {}) {
  const scope = opts.ra ? [opts.ra] : RAS;
  const seen = new Set();
  const out = [];
  for (const ra of scope) {
    const query = ra === 'Brasília' ? `${term} Brasília DF` : `${term} ${ra} Brasília DF`;
    if (seen.has(query)) continue;
    seen.add(query);
    out.push({ ra, query });
  }
  return out;
}
function buildQueries(term, opts = {}) {
  return buildPlan(term, opts).map((p) => p.query);
}
module.exports = { RAS, RAS_TIER1, RAS_TIER2, findRA, buildPlan, buildQueries, raName };
```

```js
/* Fila de detail do Capture V2: match primeiro, ambiguo ao final.
   skipPlaceIds descarta lead já detalhado nos últimos 30 dias (runner passa o Set). */
const { relevanceScore } = require('./categories');

function buildDetailQueue(rows, category, opts = {}) {
  const skip = opts.skipPlaceIds || null;
  const queue = [];
  rows.forEach((row, idx) => {
    const r = relevanceScore(row, category);
    if (r.verdict === 'reject') return;
    if (skip && row.place_id && skip.has(row.place_id)) return;
    // sem categoria no card → ambiguo: entra ao final da fila (pode receber detail se sobrar orçamento)
    const isAmbiguo = r.verdict === 'ambiguo' || !row.category;
    queue.push({ ...row, _cardIndex: idx, ...(isAmbiguo ? { _ambiguo: true } : {}) });
  });
  const rank = (x) => (x._ambiguo ? 1 : 0);
  return queue.sort((a, b) => rank(a) - rank(b) || relevanceScore(a, category).score - relevanceScore(b, category).score);
}

module.exports = { buildDetailQueue };
```

```js
async function scrapeQuery(page, query, opts = {}) {
  const t0 = Date.now();
  const url = `https://www.google.com/maps/search/${encodeURIComponent(query)}`;
  await page.goto(url, { waitUntil: 'domcontentloaded', timeout: 30000 });
  await page.getByRole('button', { name: /aceitar|accept|concord|agree/i }).click({ timeout: 3000 }).catch(() => {});
  await page.waitForSelector('a.hfpxzc, div[role="feed"]', { timeout: 20000 });
  await sleep(1500);
  const found = await scrollFeedToEnd(page, { maxScrolls: opts.maxScrolls ?? 60 });
  const cards = await extractCards(page, query);
  const relevance = opts.relevance || null; // { category, skipPlaceIds }
  const queue = relevance
    ? buildDetailQueue(cards, relevance.category, { skipPlaceIds: relevance.skipPlaceIds })
    : cards.map((c, i) => ({ ...c, _cardIndex: i }));
  const relevant = queue.filter((c) => !c._ambiguo).length;
  const ambiguous = queue.filter((c) => c._ambiguo).length;
  const dropped = cards.length - queue.length;
  const enriched = await enrichWithDetailsV2(page, queue, { maxDetails: opts.maxDetails ?? 120 });
  return {
    query,
    foundCards: found,
    leads: enriched,
    telemetry: { cards: cards.length, relevant, ambiguous, dropped, details: Math.min(queue.length, opts.maxDetails ?? 120), seconds: Math.round((Date.now() - t0) / 1000) },
  };
}

/* Detail otimizado: card el via _cardIndex; fora do limite inclui sem detail */
async function enrichWithDetailsV2(page, queue, { maxDetails = 120 } = {}) {
  const enriched = [];
  const limit = Math.min(queue.length, maxDetails);
  const els = await page.$$('a.hfpxzc').catch(() => []);
  for (let i = 0; i < limit; i++) {
    const base = queue[i];
    const card = els[base._cardIndex] || null;
    try {
      if (!card) { enriched.push(pushLead(base, false)); continue; }
      await card.scrollIntoViewIfNeeded().catch(() => {});
      await card.click({ timeout: 5000 }).catch(() => {});
      await page.waitForSelector('h1', { timeout: 6000 }).catch(() => {});
      await sleep(900);
      const d = await readPlaceDetail(page);
      enriched.push(pushLead(base, true, d));
      await page.goBack({ waitUntil: 'domcontentloaded', timeout: 8000 }).catch(() => {});
      await sleep(700);
    } catch (e) { enriched.push(pushLead(base, false)); }
    if ((i + 1) % 10 === 0) await sleep(800);
  }
  for (let i = limit; i < queue.length; i++) enriched.push(pushLead(queue[i], false));
  return enriched;
}

    lat: d.lat || base.lat || '', lng: d.lng || base.lng || '', place_id: d.place_id || base.place_id || '',
    claimed: d.claimed ?? base.claimed ?? null, hours: d.hours || base.hours || '', closed: d.closed || base.closed || false,
    photos: d.photos ?? base.photos ?? null, instagram: d.instagram || base.instagram || '',
    facebook: d.facebook || base.facebook || '', whatsapp: d.whatsapp || '',
  } : { ...base });
  if (base._ambiguo) n.ambiguo = true;
```

```js
/* Planner do Capture V2: budget proporcional, fila por yield histórico, segunda passada por termo. */
const { buildPlan } = require('./df-ra');

function defaultBudgetMinutes(nQueries) {
  // RAs demoram ~1.6 min (cards+details+jitter); DF inteiro fica capado em 60
  return Math.min(60, Math.round((6 + 1.6 * (nQueries || 0)) * 10) / 10);
}

function orderPlan(plan, history = []) {
  const byQ = new Map(history.map((h) => [h.query, h]));
  if (!history.length) return plan.slice();
  return plan.slice().sort((a, b) => {
    const ya = byQ.get(a.query);
    const yb = byQ.get(b.query);
    const ra = ya ? (ya.relevant || 0) : 1; // sem histórico = prioridade média
    const rb = yb ? (yb.relevant || 0) : 1;
    return rb - ra;
  });
}

function lowYieldRAs(plan, telemetry = [], { threshold = 3, termsPool = [] } = {}) {
  const tel = new Map(telemetry.map((t) => [t.query, t]));
  const low = plan.filter((p) => (tel.get(p.query)?.relevant || 0) < threshold);
  const out = [];
  for (const p of low) {
    for (const term of termsPool) {
      const extras = buildPlan(term, { ra: p.ra });
      out.push(...extras);
    }
  }
  return out;
}

function freshPlaceIds(rows) {
  return new Set(rows.map((r) => r.place_id).filter(Boolean));
}

module.exports = { defaultBudgetMinutes, orderPlan, lowYieldRAs, freshPlaceIds };
```

```js
  const budget = budgetMin || defaultBudgetMinutes(plan.length);
  const deadline = Date.now() + budget * 60000;

  // histórico: última meta do nicho (ordenar por yield)
  let history = [];
  const histFile = path.join('data', `${tag === 'advogados' ? 'advogados-df-max' : `${tag}-df-max`}-meta.json`);
  try { const h = JSON.parse(fs.readFileSync(histFile, 'utf8')); history = Array.isArray(h.qs) ? h.qs.map((r) => ({ query: r.query, relevant: r.relevant || 0 })) : []; } catch (e) {}

  // skip de place_id fresco (recaptura incremental)
  let skipPlaceIds = new Set();
  if (sbEnabled()) {
    try {
      const since = new Date(Date.now() - 30 * 86400000).toISOString();
      const rows = await sbFetchAll(`leads?select=place_id&source=eq.${encodeURIComponent(tag)}&updated_at=gte.${since}`);
      skipPlaceIds = freshPlaceIds(rows);
    } catch (e) {}
  }

  const ordered = orderPlan(plan, history);
  console.log(`[capture] ${ordered.length} buscas | budget=${budget} min | skipPlaceIds=${skipPlaceIds.size}`);

  const t0 = Date.now();
  const opts = { maxDetails, maxScrolls, relevance: { category: tag, skipPlaceIds } };
  const p1 = await runQueue(ordered, opts, deadline);
  const allTel = [...p1.telemetry];
  let allLeads = p1.leads;
  let cutByBudget = p1.partial;

  // 2ª passada: termos extras nas RAs de baixo yield, dentro do budget
  const t2Pool = termsOfCategory(tag).slice(1);
  if (t2Pool.length && Date.now() < deadline - QUERY_MS_MARGIN) {
    const nonExplicit = ordered.filter((p) => p.ra);
    const second = lowYieldRAs(nonExplicit, allTel, { threshold: 3, termsPool: t2Pool });
    if (second.length) {
      console.log(`[capture] 2ª passada: ${second.length} buscas (${t2Pool.join(', ')})`);
      const p2 = await runQueue(second, opts, deadline);
      allLeads = allLeads.concat(p2.leads);
      allTel.push(...p2.telemetry);
      cutByBudget = cutByBudget || p2.partial;
    }
  }

  const uniq = dedupeLeads(allLeads.map((l) => ({ ...l, name: l.name || '', address: l.address || '' })).filter((l) => l.name));
  const { kept, dropped } = relevanceFilter(uniq, tag); // defesa extra antes de salvar
  const stamp = new Date().toISOString().slice(0, 10);
  const out = await saveLeads(kept, { basename, outDir: './data', formats: ['csv', 'json', 'xlsx'] });

  const meta = {
    generated_at: new Date().toISOString(), category: tag,
    budget_min: budget, ra_count: ordered.length,
    qs: allTel, rejected_sample: dropped.slice(0, 15).map((r) => ({ name: r.name, category: r.category })),
  };
  fs.writeFileSync(path.join('data', `${basename}-meta.json`), JSON.stringify(meta, null, 2), 'utf8');

  console.log(`[capture] RAW=${allLeads.length} KEPT=${kept.length} DROPPED=${dropped.length} em ${Math.round((Date.now() - t0) / 1000)}s`);
  console.log(JSON.stringify(out, null, 2));
  if (sbEnabled()) {
    try {
      const s = await syncLeads({
        category: tag, runBuild: true,
          capture: { category: tag, qs: allTel, budget_min: budget, ra_count: ordered.length, partial: cutByBudget },
      });
      console.log(`[sync] ${s.synced} leads → Supabase (${s.file})`);
    } catch (e) {
      console.error('[sync] falhou (captura salva localmente):', e.message);
    }
  }
}

async function runQueue(queue, opts, deadline) {
  const leads = [];
  const telemetry = [];
  let partial = false;
  for (let i = 0; i < queue.length; i++) {
    if (i > 0 && Date.now() > deadline - QUERY_MS_MARGIN) {
      console.log(`[capture] budget estourado: ${queue.length - i} buscas cortadas`);
      partial = true;
      break;
    }
    const sub = await scrapeMany([queue[i].query], opts);
    leads.push(...sub.leads);
    telemetry.push(...sub.telemetry.map((t) => ({ ...t, ra: queue[i].ra })));
  }
  return { leads, telemetry, partial };
}

if (require.main === module) {
  main().catch((e) => {
    console.error(e);
    process.exit(1);
  });
}

module.exports = { main };
```
